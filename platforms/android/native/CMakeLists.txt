cmake_minimum_required(VERSION 3.22)
project(PinballAndroid LANGUAGES C CXX)
option(PINBALL_AUDIO_DIAGNOSTICS "Enable bounded local main-library audio diagnostics" OFF)
foreach(dependency SDL2 SDL2_mixer)
    if(dependency STREQUAL "SDL2")
        set(version "2.30.12")
    else()
        set(version "2.8.1")
    endif()
    set(source "${CMAKE_CURRENT_LIST_DIR}/../vendor/${dependency}-${version}")
    if(EXISTS "${source}/CMakeLists.txt")
        string(TOUPPER "${dependency}" key)
        set("FETCHCONTENT_SOURCE_DIR_${key}" "${source}" CACHE PATH "Verified local release sources" FORCE)
    endif()
endforeach()
add_subdirectory("${CMAKE_CURRENT_LIST_DIR}/../../../engine" "${CMAKE_CURRENT_BINARY_DIR}/engine")
target_sources(main PRIVATE "${CMAKE_CURRENT_LIST_DIR}/pinball_jni.cpp")
if(PINBALL_AUDIO_DIAGNOSTICS)
    target_compile_definitions(main PRIVATE PINBALL_AUDIO_DIAGNOSTICS=1)
    # Shared game sources are compiled as OBJECT files before they are linked into main.
    target_compile_definitions(pinball_core PRIVATE PINBALL_AUDIO_DIAGNOSTICS=1)
endif()
target_include_directories(main PRIVATE "${CMAKE_CURRENT_LIST_DIR}/../../../engine/SpaceCadetPinball")
target_link_options(main PRIVATE "-Wl,-z,max-page-size=16384" "-Wl,-z,common-page-size=16384")
